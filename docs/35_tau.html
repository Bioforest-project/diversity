data <- read_tsv("../modelling/data/derived_data/data_rec.tsv",
  col_types = cols()
) %>%
  select(-y) %>%
  filter(variable %in% c("ba", "diversity_q1_gen")) %>%
  select(-sitenum, -plotnum) %>%
  pivot_wider(names_from = variable, values_from = value) %>%
  na.omit() %>%
  filter(!(site %in% c(
    "Manare", "Montagne Tortue", "Nelliyampathy",
    "Uppangala", "BAFOG", "Sao Nicolau",
    "Kabo", "Mil", "Corinto",
    "Peixoto", "Iwokrama", "Antimary", "Peteco"
  ))) %>%
  mutate(sitenum = as.numeric(as.factor(site))) %>%
  mutate(plotnum = as.numeric(as.factor(paste(site, plot)))) %>%
  arrange(sitenum, plotnum, rel_year) %>%
  gather(
    variable, y, -plot, -year, -site, -treatment, -harvest_year,
    -longitude, -harvest_year_min, -rel_year, -sitenum, -plotnum
  )
metrics <- data %>%
  filter(variable == "diversity_q1_gen") %>%
  group_by(site, plot) %>%
  summarise(
    censuses = n(),
    duration = max(year) - min(year) + 1
  ) %>%
  mutate(frequency = duration / censuses) %>%
  group_by(site) %>%
  summarise(
    censuses = median(censuses),
    duration = median(duration),
    frequency = median(frequency)
  )
parameters <- read_tsv("chains/joint-diversity_q1_gen-parameters.tsv") %>%
  left_join(metrics)

parameters %>% 
  filter(variable == "tau0") %>%
  ggplot(aes(median, reorder(site, median), col = duration)) +
  geom_vline(xintercept = -3, col = "darkgrey") +
  geom_segment(aes(
    x = (median - sd),
    xend = (median + sd)
  )) +
  geom_segment(aes(x = q5, xend = q95), size = .1) +
  geom_point() +
  geom_text(aes(label = paste(round(duration, 1), "yr")), nudge_y = .5, size = 2) +
  theme_bw() +
  scale_x_continuous(labels = ~ . + 3) +
  xlab(expression(tau ~ "[ yr ]")) + ylab("") +
  ggtitle("Duration [ yr ]") +
  scale_color_viridis_c(guide = "none")

sim_sqrt_trans <- function() {
  trans_new(
    "sim_sqrt",
    function(x) sign(x) * sqrt(abs(x)),
    function(x) sign(x) * x^2
  )
}
delta_mean <- parameters %>% 
  filter(variable == "delta") %>% 
  group_by(site) %>% 
  select(median) %>% 
  summarise_all(median) %>% 
  mutate(delta_mean = median*100) %>% 
  select(site, delta_mean)
