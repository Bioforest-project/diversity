dif <- 0
var <- "diversity_q0_gen"
model <- cmdstan_model("models/disturbance_plot.stan")
basename <- paste0("distplot-", var)
out_dir <- file.path("chains", basename)

data <- read_tsv("../modelling/data/derived_data/data.tsv",
    col_types = cols()
  ) %>%
  select(-y) %>% 
  filter(treatment == "logged") %>% 
  filter(variable %in% c("ba", var)) %>% 
  na.omit() %>%
  filter(!(site %in% c(
      "Manare", "Montagne Tortue", "Nelliyampathy",
      "Uppangala", "BAFOG", "Sao Nicolau",
      "Kabo", "Mil", "Corinto",
      "Peixoto", "Iwokrama", "Antimary", "Peteco"
  ))) %>% 
  select(-sitenum) %>% 
  pivot_wider(names_from = variable, values_from = value)
data_u <- filter(data, rel_year < 0)
data_r <- data_u %>% 
  select(site, plot) %>% 
  unique() %>% 
  left_join(data) %>% 
  filter(rel_year > dif)
dist_obs <- data_r %>%
    group_by(site, plot) %>%
    summarise(ba_post = min(ba, na.rm = TRUE)) %>%
    left_join(data_u %>% 
      group_by(site, plot) %>%
      summarise(ba_pre = median(ba, na.rm = TRUE))) %>%
    mutate(dist_obs = (ba_pre - ba_post) / ba_pre) %>%
    select(site, plot, dist_obs) %>% 
  mutate(dist_obs = ifelse(dist_obs < 0, 0, dist_obs))
ind <- data_r %>%
    select(site, plot) %>%
    unique() %>% 
  left_join(dist_obs) %>% 
    mutate(sitenum = as.numeric(as.factor(site))) %>%
    mutate(plotnum = as.numeric(as.factor(paste(site, plot))))
data_u <- ind %>% 
  left_join(data_u)
data_r <- ind %>% 
  left_join(data_r)
mdata <- list(
  n = nrow(data_r),
  s = max(data_r$sitenum),
  p = max(data_r$plotnum),
  y = data_r$diversity_q0_gen,
  t = data_r$rel_year-dif,
  site = data_r$sitenum,
  plot = data_r$plotnum,
  site_plot = ind$sitenum,
  n_u = nrow(data_u),
  y_u = data_u$diversity_q0_gen,
  plot_u = data_u$plotnum,
  dist_p = ind$dist_obs
)

fit <- model$sample(
    mdata,
    chains = 4,
    parallel_chains = 4,
    save_warmup = FALSE,
    refresh = 100,
    output_dir = out_dir
)
